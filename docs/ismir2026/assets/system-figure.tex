% TikZ system pipeline — Pocket Producer
% Two-lane design: cool lane = system proposes, warm lane = musician decides.
% Requires tikz libraries: arrows.meta, positioning, fit, backgrounds.
\definecolor{csysline}{HTML}{4A7FB5}
\definecolor{csysfill}{HTML}{EDF3FA}
\definecolor{cusrline}{HTML}{C0863C}
\definecolor{cusrfill}{HTML}{FBF3E4}
\definecolor{cpanelsys}{HTML}{F7FAFD}
\definecolor{cpanelusr}{HTML}{FEFBF5}
\definecolor{cedge}{HTML}{44484D}
\definecolor{ctext}{HTML}{20242A}
\definecolor{csub}{HTML}{6B7077}

\begin{tikzpicture}[
    node distance=0.42cm and 0.5cm,
    box/.style 2 args={
        rectangle, rounded corners=3pt,
        minimum width=2.4cm, minimum height=0.95cm,
        draw=#1, line width=0.7pt, fill=#2,
        text=ctext, align=center, inner sep=3pt,
        font=\sffamily\footnotesize,
    },
    badge/.style={
        circle, fill=#1, text=white,
        font=\sffamily\tiny\bfseries,
        inner sep=0pt, minimum size=10.5pt,
    },
    arr/.style={
        -{Stealth[length=4.5pt, width=4pt]},
        line width=0.6pt, color=cedge,
    },
    fbarr/.style={
        -{Stealth[length=4.5pt, width=4pt]},
        line width=0.6pt, color=csysline,
        dash pattern=on 2.6pt off 2pt,
    },
]

% ---- Lane 1: system proposes ----
\node[box={csysline}{csysfill}] (S)
    {\textbf{DAW Session}\\[-1pt]
     {\tiny\color{csub}Audiotool Nexus SDK}};
\node[box={csysline}{csysfill}, right=of S] (F)
    {\textbf{Fingerprint}\\[-1pt]
     {\tiny\color{csub}tempo $\cdot$ CLAP embeddings}};
\node[box={csysline}{csysfill}, right=of F] (R)
    {\textbf{Ranking}\\[-1pt]
     {\tiny\color{csub}5 signals $\cdot$ BPR fusion}};

% ---- Lane 2: musician decides ----
\node[box={cusrline}{cusrfill}, below=1.05cm of R] (P)
    {\textbf{Preview}\\[-1pt]
     {\tiny\color{csub}per-signal evidence}};
\node[box={cusrline}{cusrfill}, left=of P] (A)
    {\textbf{Insert / Undo}\\[-1pt]
     {\tiny\color{csub}one Nexus transaction}};
\node[box={cusrline}{cusrfill}, left=of A] (L)
    {\textbf{Feedback Log}\\[-1pt]
     {\tiny\color{csub}accept $\cdot$ reject $\cdot$ undo}};

% ---- Lane panels (background) ----
\begin{scope}[on background layer]
\node[rectangle, rounded corners=4.5pt, fill=cpanelsys,
      draw=csysline!30, line width=0.4pt,
      fit=(S)(R), inner sep=7pt] (PSYS) {};
\node[rectangle, rounded corners=4.5pt, fill=cpanelusr,
      draw=cusrline!35, line width=0.4pt,
      fit=(P)(L), inner sep=7pt] (PUSR) {};
\end{scope}

% ---- Lane labels ----
\node[font=\sffamily\tiny\bfseries, text=csysline, anchor=south west]
    at ([yshift=1.5pt]PSYS.north west) {SYSTEM PROPOSES};
\node[font=\sffamily\tiny\bfseries, text=cusrline, anchor=north west]
    at ([yshift=-1.5pt]PUSR.south west) {MUSICIAN DECIDES};

% ---- Step badges ----
\node[badge=csysline] at (S.north west) {1};
\node[badge=csysline] at (F.north west) {2};
\node[badge=csysline] at (R.north west) {3};
\node[badge=cusrline] at (P.north west) {4};
\node[badge=cusrline] at (A.north west) {5};
\node[badge=cusrline] at (L.north west) {6};

% ---- Forward flow ----
\draw[arr] (S) -- (F);
\draw[arr] (F) -- (R);
\draw[arr] (R.south) -- (P.north);
\draw[arr] (P) -- (A);
\draw[arr] (A) -- (L);

% ---- Feedback loop ----
\coordinate (fb) at ([xshift=-0.6cm]L.west);
\draw[fbarr] (L.west) -- (fb) |- (S.west);
\node[font=\sffamily\tiny\itshape, text=csysline, rotate=90, anchor=south]
    at ([shift={(-0.08,1.0)}]fb) {future personalization};

\end{tikzpicture}
